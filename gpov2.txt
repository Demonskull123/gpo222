local CONNS = {}
local function track(c) if c then CONNS[#CONNS+1] = c end return c end
local function untrack(c)
    if not c then return end
    for i = #CONNS, 1, -1 do if CONNS[i] == c then table.remove(CONNS, i) break end end
    pcall(function() c:Disconnect() end)
end
local function clearConns()
    for _, c in ipairs(CONNS) do pcall(function() c:Disconnect() end) end
    table.clear(CONNS)
end

local RemoteCache = {}

local function findRemote(name)
    if RemoteCache[name] then
        local r = RemoteCache[name]
        if r and r.Parent then return r end
        RemoteCache[name] = nil
    end
    local candidates = RemoteNames[name] or {name}
    for _, n in ipairs(candidates) do
        local r = Net:FindFirstChild(n)
            or RemoteEvents:FindFirstChild(n)
            or RemoteFunctions:FindFirstChild(n)
            or Remotes:FindFirstChild(n)
            or Comm and Comm:FindFirstChild(n)
        if r and (r:IsA("RemoteEvent") or r:IsA("RemoteFunction")) then
            RemoteCache[name] = r
            return r
        end
    end
end

local function fireRemote(name, ...)
    local r = findRemote(name)
    if not r then return false end
    pcall(function() r:FireServer(...) end)
    return true
end

local function invokeRemote(name, ...)
    local r = findRemote(name)
    if not r or not r:IsA("RemoteFunction") then return nil end
    local ok, res = pcall(function() return r:InvokeServer(...) end)
    if ok then return res end
end

local function distanceOf(a, b) if not a or not b then return math.huge end return (a-b).Magnitude end

local function distance2D(a, b)
    if not a or not b then return math.huge end
    local dx, dz = a.X-b.X, a.Z-b.Z
    return math.sqrt(dx*dx + dz*dz)
end

local function angleTo(a, b)
    if not a or not b then return 0 end
    return math.atan2(b.Z - a.Z, b.X - a.X)
end

local function lerpNum(a, b, t) return a + (b-a)*t end
local function clampNum(v, lo, hi) if v < lo then return lo end if v > hi then return hi end return v end
local function roundNum(v) return math.floor(v + 0.5) end

local function splitString(s, sep)
    local out = {}
    for part in string.gmatch(s, "([^" .. sep .. "]+)") do out[#out+1] = part end
    return out
end

local function trimString(s) return (s:gsub("^%s*(.-)%s*$", "%1")) end
local function lowerString(s) return string.lower(s or "") end
local function upperString(s) return string.upper(s or "") end

local function containsIgnoreCase(h, n)
    if not h or not n then return false end
    return string.find(string.lower(h), string.lower(n), 1, true) ~= nil
end

local function startsWith(s, p) if not s or not p then return false end return string.sub(s, 1, #p) == p end
local function endsWith(s, p) if not s or not p then return false end return string.sub(s, -#p) == p end

local function tableContains(t, v)
    if not t then return false end
    for _, x in ipairs(t) do if x == v then return true end end
    return false
end

local function tableCopy(t)
    local out = {}
    if not t then return out end
    for k, v in pairs(t) do out[k] = v end
    return out
end

local function tableMerge(a, b)
    local out = tableCopy(a)
    if not b then return out end
    for k, v in pairs(b) do out[k] = v end
    return out
end

local function tableKeys(t)
    local out = {}
    if not t then return out end
    for k in pairs(t) do out[#out+1] = k end
    return out
end

    for _ in pairs(t) do n = n + 1 end
    return n

    if not t then return nil end
    for i, x in ipairs(t) do if x == v then return i end end
end

local function tableRemoveValue(t, v)
    if not t then return end
    for i = #t, 1, -1 do if t[i] == v then table.remove(t, i) end end
end

local function tableShuffle(t)
    if not t then return t end
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
    return t
end

local function tableUnique(t)
    local seen, out = {}, {}
    if not t then return out end
    for _, v in ipairs(t) do
        if not seen[v] then seen[v] = true; out[#out+1] = v end
    end
    return out
end

    placeId = placeId or game.PlaceId
    pcall(function()
        local url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", placeId)

        if not data or not data.data then return end
        local best = nil
        for _, s in ipairs(data.data) do
            if s.playing < s.maxPlayers and s.id ~= game.JobId then
                if not best or (s.ping and best.ping and s.ping < best.ping) then best = s end
            end
        end
        if best then TeleportService:TeleportToPlaceInstance(placeId, best.id, LocalPlayer) end
    end)

        debug.getinfo = function(threadOrLevel, what, ...)
            if type(threadOrLevel) == "number" then

                    if info.source:find("@") == nil then

    for _, tag in ipairs({"boss","raid","kraken","seabeast","seaking","titan","awakened"}) do

local function forAllModels(fn)
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("Model") then pcall(fn, d) end
    end
end

function Entity.nearest(filter, maxDist, anchor)
    local a = anchor or myRoot()
    if not a then return nil, math.huge end
    local best, bd = nil, maxDist or math.huge
    forAllModels(function(m)
        if filter(m) then
            local p = Entity.pos(m)
            if p then
                local d = (p - a.Position).Magnitude
                if d < bd then best, bd = m, d end
            end
        end
    end)
    return best, bd
end

function Entity.all(filter, maxDist, anchor)
    local a = anchor or myRoot()
    if not a then return {} end
    local out = {}
    forAllModels(function(m)
        if filter(m) then
            local p = Entity.pos(m)
            if p then
                local d = (p - a.Position).Magnitude
                if not maxDist or d <= maxDist then
                    out[#out+1] = {model=m, dist=d, pos=p}
                end
            end
        end
    end)
    table.sort(out, function(x,y) return x.dist < y.dist end)
    return out
end

function Entity.lowestHP(list)
    local best, bh = nil, math.huge
    for _, e in ipairs(list) do
        local h = Entity.hum(e.model)
        if h and h.Health < bh then best, bh = e.model, h.Health end
    end
    return best
end

function Entity.highestHP(list)
    local best, bh = nil, -1
    for _, e in ipairs(list) do
        local h = Entity.hum(e.model)
        if h and h.Health > bh then best, bh = e.model, h.Health end
    end
    return best
end

function Entity.highestLevel(list)
    local best, bl = nil, -1
    for _, e in ipairs(list) do
        local lv = e.model:GetAttribute("Level") or 0
        if lv > bl then best, bl = e.model, lv end
    end
    return best
end

function Entity.pickByMode(mode, filter, maxDist)
    local list = Entity.all(filter, maxDist)
    if #list == 0 then return nil end
    if mode == "Lowest HP" then return Entity.lowestHP(list) end
    if mode == "Highest HP" then return Entity.highestHP(list) end
    if mode == "Highest Level" then return Entity.highestLevel(list) end
    return list[1].model
end

function Entity.faceTowards(target)
    local r = myRoot()
    local p = Entity.pos(target)
    if not (r and p) then return end
    pcall(function()
        r.CFrame = CFrame.lookAt(r.Position, Vector3.new(p.X, r.Position.Y, p.Z))
    end)
end

function Entity.moveTo(target, offset)
    local p = Entity.pos(target)
    if not p then return end
    teleportTo(p + Vector3.new(0, offset or 3, 0))
end

function Entity.moveBehind(target, dist)
    local r = myRoot()
    local p = Entity.pos(target)
    if not (r and p) then return end
    local dir = (r.Position - p).Unit
    teleportTo(p + dir * (dist or 4))
end

function Entity.distanceTo(target)
    local r = myRoot()
    local p = Entity.pos(target)
    if not (r and p) then return math.huge end
    return (r.Position - p).Magnitude
end

function Entity.aimAt(target)
    local r = myRoot()
    local p = Entity.pos(target)
    if not (r and p) then return end
    local hum = myHum()
    if hum then pcall(function() hum.AutoRotate = false end) end
    pcall(function() r.CFrame = CFrame.lookAt(r.Position, p) end)
end

local function setFly(on)
    if Movement.fly then Movement.fly:Disconnect() Movement.fly = nil end
    if Movement.flyBV then pcall(function() Movement.flyBV:Destroy() end) Movement.flyBV = nil end
    if Movement.flyBG then pcall(function() Movement.flyBG:Destroy() end) Movement.flyBG = nil end
    if not on then return end
    local r = myRoot()
    if not r then return end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Velocity = Vector3.zero
    bv.Parent = r
    Movement.flyBV = bv
    local bg = Instance.new("BodyGyro")
    bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bg.P = 12000
    bg.Parent = r
    Movement.flyBG = bg
    local cam = Workspace.CurrentCamera
    Movement.fly = RunService.RenderStepped:Connect(function()
        if not isAlive() then return end
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
        bv.Velocity = dir.Magnitude > 0 and dir.Unit * Config.FlySpeed or Vector3.zero
        bg.CFrame = cam.CFrame
    end)
end

local function setAutoDodge(on)
    if Movement.dodge then Movement.dodge:Disconnect() Movement.dodge = nil end
    if not on then return end
    local lastDodge = 0
    Movement.dodge = RunService.Heartbeat:Connect(function()
        local now = os.clock()
        if now - lastDodge < 0.4 then return end
        local r = myRoot()
        if not r then return end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hrp and (hrp.Position - r.Position).Magnitude <= Config.DodgeRange then
                    local dir = (r.Position - hrp.Position).Unit
                    r.CFrame = r.CFrame + dir * 12

local function clearESP()
    for _, v in pairs(ESP.cache) do pcall(function() v:Destroy() end) end
    table.clear(ESP.cache)
    for _, v in pairs(ESP.healthCache) do pcall(function() v:Destroy() end) end
    table.clear(ESP.healthCache)
    for _, v in pairs(ESP.nameCache) do pcall(function() v:Destroy() end) end
    table.clear(ESP.nameCache)
    for _, v in pairs(ESP.distanceCache) do pcall(function() v:Destroy() end) end
    table.clear(ESP.distanceCache)
    if ESP.folder then
        pcall(function() ESP.folder:Destroy() end)
        ESP.folder = nil
    end
end

    for _, v in pairs(ESP.tracerCache) do pcall(function() v:Destroy() end) end

        pcall(function() ESP.tracerFolder:Destroy() end)
        ESP.tracerFolder = nil
    end
end

local function drawESP(inst, color, name)
    if ESP.cache[inst] then
        pcall(function()
            ESP.cache[inst].FillColor = color
            ESP.cache[inst].OutlineColor = color
        end)
        return ESP.cache[inst]
    end
    local hl = Instance.new("Highlight")
    hl.Name = name or "TempoHL"
    hl.FillColor = color
    hl.FillTransparency = 0.55
    hl.OutlineColor = color
    hl.OutlineTransparency = 0.15

    hl.Parent = ensureESPFolder()

    return hl
end

local function espStep()
    if not HubAlive then return end
    local any = Persist.ESPPlayers or Persist.ESPFruits or Persist.ESPChests
        or Persist.ESPBosses or Persist.ESPNPCs or Persist.ESPResources
        or Persist.ESPShips or Persist.ESPPets
    if not any then
        if next(ESP.cache) then clearESP() end
        if next(ESP.tracerCache) then clearTracers() end
        return
    end
    local r = myRoot()
    if not r then return end
    local range = Config.ESPRange
    local fruitRange = Config.ESPFruitRange
    local chestRange = Config.ESPChestRange

    if Persist.ESPPlayers then
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hrp and (hrp.Position - r.Position).Magnitude <= range then
                    drawESP(p.Character, Config.ESPColorPlayer, "ESP_Player")
                    if Persist.ESPTracers then drawTracer(p.Character, Config.ESPColorTracer) end
                    if Persist.ESPHealth then drawHealth(p.Character, Config.ESPColorPlayer) end
                    if Persist.ESPPlayerNames then
                        local plr = Players:GetPlayerFromCharacter(p.Character)
                        if plr then drawName(p.Character, plr.Name, Config.ESPColorPlayer) end
                    end
                    if Persist.ESPDistance then
                        local d = math.floor((hrp.Position - r.Position).Magnitude)
                        drawDistance(p.Character, tostring(d) .. "m", Config.ESPColorPlayer)
                    end
                end
            end
        end
    end

    for _, m in ipairs(Workspace:GetDescendants()) do
        if m:IsA("Model") then
            local root = Entity.root(m)
            if root then
                local dist = (root.Position - r.Position).Magnitude
                if Persist.ESPBosses and dist <= range and Entity.isBoss(m) then
                    drawESP(m, Config.ESPColorBoss, "ESP_Boss")
                    if Persist.ESPHealth then drawHealth(m, Config.ESPColorBoss) end
                    if Persist.ESPDistance then
                        drawDistance(m, tostring(math.floor(dist)) .. "m", Config.ESPColorBoss)
                    end
                elseif Persist.ESPChests and dist <= chestRange and Entity.isChest(m) then
                    drawESP(m, Config.ESPColorChest, "ESP_Chest")
                elseif Persist.ESPFruits and dist <= fruitRange and Entity.isFruit(m) then
                    drawESP(m, Config.ESPColorFruit, "ESP_Fruit")
                    if Persist.ESPFruitNames then drawName(m, m.Name, Config.ESPColorFruit) end
                elseif Persist.ESPShips and dist <= range and Entity.isShip(m) then
                    drawESP(m, Config.ESPColorShip, "ESP_Ship")
                elseif Persist.ESPResources and dist <= range and Entity.isResource(m) then
                    drawESP(m, Config.ESPColorResource, "ESP_Resource")
                elseif Persist.ESPPets and dist <= range and Entity.isPet(m) then
                    drawESP(m, Color3.fromRGB(255, 180, 240), "ESP_Pet")
                elseif Persist.ESPNPCs and dist <= range and Entity.isNPC(m) and not Entity.isBoss(m) then
                    drawESP(m, Config.ESPColorNPC, "ESP_NPC")
                    if Persist.ESPHealth then drawHealth(m, Config.ESPColorNPC) end
                end
            end
        end
    end
end

local function questStep()
    if not Persist.AutoQuest then return end
    if os.clock() - Farm.lastQuest < 2 then return end
    Farm.lastQuest = os.clock()
    fireRemote("Quest", "Accept")
    fireRemote("QuestAccept")
    bumpCounter("quests")
end

local function autoFishStep()
    if not Persist.AutoFish then return end
    if os.clock() - Farm.lastFish < 2.5 then return end
    Farm.lastFish = os.clock()
    fireRemote("Fish", "Cast")
    task.wait(Config.FishReelDelay or 0.35)
    if Config.FishMode == "Perfect" then
        for _ = 1, 6 do
            if not Persist.AutoFish then return end
            fireRemote("Fish", "Reel")
            task.wait(0.12)
        end
    else
        fireRemote("Fish", "Reel")
    end
end

local function autoSellStep()
    if not Persist.AutoSell then return end
    if os.clock() - Farm.lastSell < 5 then return end
    Farm.lastSell = os.clock()
    fireRemote("Sell", "All")
    fireRemote("FishMarket", "SellAll")
end

local function seaBeastStep()
    if not (Persist.AutoSeaBeast or Persist.AutoKraken) then return end
    local range = Persist.AutoKraken and Config.KrakenRange or Config.SeabeastRange
    range = range or Config.SeabeastRange
    local t = Entity.nearest(function(m)
        local n = string.lower(m.Name)
        return string.find(n, "seabeast", 1, true)
            or string.find(n, "seaking", 1, true)
            or (Persist.AutoKraken and string.find(n, "kraken", 1, true))
    end, range)
    if not t then return end
    Entity.faceTowards(t)
    for i = 1, 5 do useSkill(i) end
    useWeapon()
    bumpCounter("seaBeasts")
end

local function autoShipStep()
    if not Persist.AutoShip then return end
    if os.clock() - Farm.lastShip < 3 then return end
    Farm.lastShip = os.clock()
    local ship = Entity.nearest(function(m)
        local n = string.lower(m.Name)
        return string.find(n, "ship", 1, true)
            or string.find(n, "boat", 1, true)
            or string.find(n, "vessel", 1, true)
    end, 400)
    if ship then Entity.moveTo(ship, 8) end
end

local function autoBossStep()
    if not Persist.AutoBoss then return end
    local b = Entity.pickByMode(Config.BossMode, Entity.isBoss, 900)
    if not b then return end
    Farm.target = b
    if Entity.distanceTo(b) > 18 then Entity.moveTo(b, 3) end
    Entity.faceTowards(b)
    for i = 1, 5 do useSkill(i) end
    useWeapon()
    bumpCounter("bosses")
end

local function weaponRotateStep()
    local now = os.clock()
    if now - Farm.weaponRotate < 0.6 then return end
    Farm.weaponRotate = now
    local rotate = Config.SwordSkillRotate
    if #rotate > 0 then
        Farm.skillRotate = (Farm.skillRotate % #rotate) + 1
        useSkill(rotate[Farm.skillRotate])
    end
end

local function farmStep()
    if not Farm.running or not HubAlive then return end
    if not isAlive() then
        bumpCounter("deaths")
        return
    end
    local now = os.clock()
    if now - Farm.lastAttack < 0.12 / math.max(Config.FarmSpeed, 0.05) then return end
    Farm.lastAttack = now

    questStep()

    if Persist.AutoBoss then
        autoBossStep()
        return
    end

    if Persist.AutoFarm then
        local n = Entity.pickByMode("Nearest", Entity.isNPC, 400)
        if n and not Entity.isBoss(n) then
            Farm.target = n
            if Entity.distanceTo(n) > 16 then Entity.moveTo(n, 3) end
            Entity.faceTowards(n)
            for i = 1, 5 do useSkill(i) end
            useWeapon()
            weaponRotateStep()
            bumpCounter("kills")
            return
        end
    end

    if Persist.AutoChest then
        local c = Entity.pickByMode("Nearest", Entity.isChest, Config.ChestRange)
        if c then
            if Entity.distanceTo(c) > 6 then
                Entity.moveTo(c, 3)
            else
                if now - Farm.lastChestOpen > 0.5 then
                    Farm.lastChestOpen = now
                    fireRemote("Chest", "Open")
                    fireRemote("OpenChest", c)
                    bumpCounter("chests")
                end
            end
            return
        end
    end

    seaBeastStep()
    autoShipStep()
    autoSellStep()
    autoFishStep()
end

local function startFarm()
    if Farm.running then return end
    Farm.running = true
    Farm.conn = RunService.Heartbeat:Connect(function() pcall(farmStep) end)
    track(Farm.conn)
end

local function stopFarm()
    Farm.running = false
    if Farm.conn then
        Farm.conn:Disconnect()
        Farm.conn = nil
    end
end

    Kaitun.lastAccept = os.clock()
    fireRemote("Quest", "Accept")
    fireRemote("QuestAccept")
    bumpCounter("quests")
end

local function kaitunTravelTo(targetIsland)
    if not targetIsland then return end
    local pos = KaitunIslandPositions[targetIsland]
    if not pos then return end
    local r = myRoot()
    if not r then return end
    if (r.Position - pos).Magnitude <= 30 then return end
    if os.clock() - Kaitun.lastTravel < 2 then return end
    Kaitun.lastTravel = os.clock()
    teleportTo(pos)
end

local function kaitunAttackNearest(range)
    local r = myRoot()
    if not r then return end
    local n = Entity.pickByMode("Nearest", Entity.isNPC, range or 400)
    if not n then return end
    if Entity.distanceTo(n) > 16 then Entity.moveTo(n, 3) end
    Entity.faceTowards(n)
    for i = 1, 5 do useSkill(i) end
    useWeapon()
    if Config.AutoKaitunSwordSkill and os.clock() - Kaitun.weaponSwap > 0.5 then
        Kaitun.weaponSwap = os.clock()
        useSword()
    end
    if Config.AutoKaitunGunSkill then useGun() end
    if Config.AutoKaitunFruitSkill then
        local fr = Config.FruitSkillRotate
        if fr and #fr > 0 then useSkill(fr[math.random(#fr)]) end
    end
    if Config.AutoKaitunStyleSkill then useStyle() end
    bumpCounter("kills")
end

local function kaitunStep()
    if not Kaitun.running or not HubAlive then return end
    if not isAlive() then
        bumpCounter("deaths")
        return
    end
    local now = os.clock()
    if now - Kaitun.lastStep < 0.15 then return end
    Kaitun.lastStep = now

    local q = GPO.quest()
    local done, goal = GPO.questProgress()

    if not q then
        kaitunAcceptQuest()
        return
    end

    if kaitunIsTargetQuest(q) then
        kaitunAttackNearest(400)
        return
    end

    if done and goal and done >= goal then
        if now - Kaitun.lastComplete > 1.5 then
            Kaitun.lastComplete = now
            fireRemote("Quest", "Complete")
            fireRemote("QuestComplete")
            Kaitun.cycles = Kaitun.cycles + 1
        end
        return
    end

    if Config.KaitunIsland and Config.KaitunIsland ~= "" then
        kaitunTravelTo(Config.KaitunIsland)
    end
end

local function startKaitun()
    if Kaitun.running then return end
    Kaitun.running = true
    Kaitun.conn = RunService.Heartbeat:Connect(function() pcall(kaitunStep) end)
    track(Kaitun.conn)
end

local function stopKaitun()
    Kaitun.running = false
    if Kaitun.conn then
        Kaitun.conn:Disconnect()
        Kaitun.conn = nil
    end
end

    if Config.ComboMode ~= "Auto" then return end
    local now = os.clock()
    if now - comboLast < Config.ComboDelay then return end
    comboLast = now
    local r = myRoot()
    if not r then return end
    local target = Entity.nearest(function(m)
        if not Entity.isPlayer(m) then return false end
        local h = Entity.hum(m)
        return h and h.Health > 0
    end, Config.AuraRange)

    Entity.faceTowards(target)
    comboIndex = (comboIndex % 5) + 1
    useSkill(comboIndex)

local Pathing = {
    waypoint = nil,
    conn = nil,
    mode = nil,
    steps = nil,
    delay = nil,
    active = false,
}

local function teleportThrough(pos, steps, delay)
    local r = myRoot()
    if not r then return end
    steps = steps or Config.PathingSteps
    delay = delay or Config.PathingStepDelay
    local start = r.Position
    for i = 1, steps do
        if not Pathing.active then return end
        local t = i / steps
        local lerped = start:Lerp(pos, t)
        r.CFrame = CFrame.new(lerped)
        task.wait(delay)
    end
end

local function pathStep()
    if not Pathing.waypoint then return end
    local r = myRoot()
    if not r then return end
    local d = (r.Position - Pathing.waypoint).Magnitude
    if d < 4 then
        Pathing.waypoint = nil
        Pathing.active = false
        return
    end
    local mode = Pathing.mode or Config.PathingMode
    if mode == "Direct" then
        r.CFrame = CFrame.new(Pathing.waypoint)
    elseif mode == "Lerp" then
        teleportThrough(Pathing.waypoint, Pathing.steps, Pathing.delay)
    elseif mode == "Step" then
        local dir = (Pathing.waypoint - r.Position).Unit
        r.CFrame = CFrame.new(r.Position + dir * math.min(d, 15))
    end
end

track(RunService.Heartbeat:Connect(function()
    if HubAlive and Pathing.waypoint then pcall(pathStep) end
end))

local function goTo(pos)
    Pathing.waypoint = pos
    Pathing.active = true
end

local function stopGoTo()
    Pathing.waypoint = nil
    Pathing.active = false
end

                TeleportService:TeleportToPlaceInstance(placeId, jobId, LocalPlayer)
            end)
        end
    end)
    track(Utility.rejoin)
end

local function setRejoinOnLowHealth(on)
    if Utility.rejoinHealth then Utility.rejoinHealth:Disconnect() Utility.rejoinHealth = nil end
    if not on then return end
    Utility.rejoinHealth = RunService.Heartbeat:Connect(function()
        local h = myHum()
        if h and h.Health / h.MaxHealth * 100 <= 15 then
            rejoinServer()
        end
    end)
    track(Utility.rejoinHealth)
end

local function setWebhookKills(on)
    if Utility.webhookKills then Utility.webhookKills:Disconnect() Utility.webhookKills = nil end
    if not on then return end
    if Config.WebhookUrl == "" then return end
    Utility.webhookKills = RunService.Heartbeat:Connect(function()
        local h = myHum()
        if h and h.Health <= 0 then
            pcall(function()
                local body = HttpService:JSONEncode({
                    content = "TempoHub: " .. LocalPlayer.Name .. " died",
                    username = Config.WebhookName,
                })
                HttpService:PostAsync(Config.WebhookUrl, body, Enum.HttpContentType.ApplicationJson)
            end)
        end
    end)
    track(Utility.webhookKills)
end

local function bindKey(key, fn)
    local kc = type(key) == "string" and Enum.KeyCode[key] or key
    if not kc then return end
    track(UserInputService.InputBegan:Connect(function(input, gp)
        if gp then return end
        if input.KeyCode == kc then
            pcall(fn)
        end
    end))
end

local function devTab(t)
    local button, toggle = t.Button, t.Toggle
    function t:Toggle(name, a, b)
        if type(a) == "function" then return toggle(self, name, false, a) end
        local h = toggle(self, name, a == true, b)
        if type(a) == "string" then self:Label(a) end
        return h
    end
    function t:Button(name, a, b)
        local h = button(self, name, a == true, type(a) == "function" and a or b)
        if type(a) == "string" then self:Label(a) end
        return h
    end
    function t:Textbox(name, hint, cb)
        return self:Input(name .. " - " .. tostring(hint or ""), nil, cb)
    end
    return t
end

local HitboxSystem = {

    if HitboxSystem.conn then HitboxSystem.conn:Disconnect() HitboxSystem.conn = nil end
    if not on then
        for inst, size in pairs(HitboxSystem.originalSizes) do
            pcall(function() inst.Size = size end)
        end
        table.clear(HitboxSystem.originalSizes)
        return
    end
    HitboxSystem.active = true
    HitboxSystem.conn = RunService.Heartbeat:Connect(function()
        if not HitboxSystem.active then return end
        for _, m in ipairs(Workspace:GetDescendants()) do
            if m:IsA("Model") and Entity.isNPC(m) then
                for _, part in ipairs(m:GetDescendants()) do
                    if part:IsA("BasePart") then
                        if not HitboxSystem.originalSizes[part] then
                            HitboxSystem.originalSizes[part] = part.Size
                        end
                        pcall(function()
                            part.Size = part.Size * 1.5
                        end)
                    end
                end
            end
        end
    end)
    track(HitboxSystem.conn)
end

local AntiDetection = {
    patched = false,
    originalNamecall = nil,
    originalIndex = nil,
}

local function patchMetatable()
    if AntiDetection.patched then return end
    local ok = pcall(function()
        local meta = getrawmetatable(game)
        if not meta then return end
        local origNamecall = meta.__namecall
        local origIndex = meta.__index
        AntiDetection.originalNamecall = origNamecall
        AntiDetection.originalIndex = origIndex
        setreadonly(meta, false)
        meta.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if method == "FireServer" or method == "InvokeServer" then
                if checkcaller and not checkcaller() then
                    return origNamecall(self, ...)
                end
            end
            return origNamecall(self, ...)
        end)
        meta.__index = newcclosure(function(self, key)
            return origIndex(self, key)
        end)
        setreadonly(meta, true)
    end)
    AntiDetection.patched = ok
end

pcall(patchMetatable)

local function hakiV2Step(tracker, stages)
    if not tracker.running or not HubAlive then return end
    local now = os.clock()
    if now - tracker.lastAttempt < 1.5 then return end
    tracker.lastAttempt = now
    if tracker.stage >= tracker.max then
        tracker.running = false
        return
    end
    local nextStage = stages[tracker.stage + 1]
    if not nextStage then return end
    if GPO.beli() < nextStage.cost then
        pushWebhook("Haki V2 " .. nextStage.name .. " blocked: need " .. nextStage.cost .. " beli")
        tracker.running = false
        return
    end
    fireRemote("HakiUnlock", nextStage.remote, nextStage.arg)
    fireRemote("HakiV2", nextStage.remote, nextStage.arg)
    tracker.stage = tracker.stage + 1
    pushWebhook("Haki V2 unlocked: " .. nextStage.name)
end

local PacketSpoofer = {
    installed=false,
    outgoingLog={},
    rateLimit={},
    rateWindow=0.05,
    maxPerWindow=20,
}

local function installPacketSpoofer()
    if PacketSpoofer.installed then return end
    local ok = pcall(function()
        local meta = getrawmetatable(game)
        if not meta then return end
        local origNamecall = meta.__namecall
        setreadonly(meta, false)
        meta.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if method == "FireServer" or method == "InvokeServer" then
                if checkcaller and checkcaller() then
                    local key = tostring(self)
                    local now = os.clock()
                    local entry = PacketSpoofer.rateLimit[key]
                    if not entry or now - entry.t > PacketSpoofer.rateWindow then
                        PacketSpoofer.rateLimit[key] = {t=now, n=1}
                    else
                        entry.n = entry.n + 1
                        if entry.n > PacketSpoofer.maxPerWindow then
                            return
                        end
                    end
                end
            end
            return origNamecall(self, ...)
        end)
        setreadonly(meta, true)
    end)
    PacketSpoofer.installed = ok
end

pcall(installPacketSpoofer)

local RouteOptimizer = {
    running=false,
    currentRoute={},
    currentIndex=1,
    lastAdvance=0,
}

local function buildRoute(filterFn, maxDist)
    local list = Entity.all(filterFn, maxDist or 500)
    if #list == 0 then return {} end
    local route = {list[1]}
    table.remove(list, 1)
    while #list > 0 do
        local last = route[#route].pos
        local bestIdx, bestDist = 1, math.huge
        for i, e in ipairs(list) do
            local d = (e.pos - last).Magnitude
            if d < bestDist then
                bestDist = d
                bestIdx = i
            end
        end
        route[#route+1] = list[bestIdx]
        table.remove(list, bestIdx)
    end
    return route
end

local function startRouteOptimizer()
    if RouteOptimizer.running then return end
    RouteOptimizer.running = true
    RouteOptimizer.currentRoute = buildRoute(function(m)
        return Entity.isChest(m)
    end, Config.ChestRange)
    RouteOptimizer.currentIndex = 1
    local conn = RunService.Heartbeat:Connect(function()
        if not HubAlive or not RouteOptimizer.running then return end
        local now = os.clock()
        if now - RouteOptimizer.lastAdvance < 0.5 then return end
        RouteOptimizer.lastAdvance = now
        local entry = RouteOptimizer.currentRoute[RouteOptimizer.currentIndex]
        if not entry then
            RouteOptimizer.running = false
            return
        end
        local r = myRoot()
        if not r then return end
        if (r.Position - entry.pos).Magnitude > 6 then
            teleportTo(entry.pos + Vector3.new(0, 3, 0))
        else
            fireRemote("Chest", "Open")
            fireRemote("OpenChest", entry.model)
            RouteOptimizer.currentIndex = RouteOptimizer.currentIndex + 1
        end
    end)
    track(conn)
end

local function stopRouteOptimizer()
    RouteOptimizer.running = false
    table.clear(RouteOptimizer.currentRoute)
    RouteOptimizer.currentIndex = 1
end

local function playNotificationSound(kind)
    if not AutoPlayAudio.enabled then return end
    local id = AutoPlayAudio.sounds[kind]
    if not id then return end
    playSoundEffect(id, 0.5, 1)
end

local AutoShoot = {
    running=false,
    lastShot=0,
    interval=0.5,
}

track(RunService.Heartbeat:Connect(function()
    if not HubAlive or not Persist.AutoShoot then return end
    local now = os.clock()
    if now - AutoShoot.lastShot < AutoShoot.interval then return end
    AutoShoot.lastShot = now
    local r = myRoot()
    if not r then return end
    local target = Entity.nearest(function(m)
        return Entity.isPlayer(m)
    end, 80)
    if not target then return end
    local tp = Entity.pos(target)
    if not tp then return end
    r.CFrame = CFrame.lookAt(r.Position, tp)
    fireRemote("Gun", "Shoot")
end))
